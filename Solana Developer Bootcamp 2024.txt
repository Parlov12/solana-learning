Data Types

Scalar values - single value

let a: i32 = -42;         // signed integer
let b: u32 = 42;          // unsigned integer
let c: f64 = 3.14;        // float
let d: bool = true;       // boolean
let e: char = 'z';        // character

Compung types - tuple - fixed-size group of values of different types 

let tup = (1, 3.14, 'a');      // type: (i32, f64, char)
let (x, y, z) = tup;           // destructuring
let second = tup.1;           // access by index
